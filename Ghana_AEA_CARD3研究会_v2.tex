\begin{frame}{Motivation}
	\begin{itemize}
		\item To increase agricultural productivity, adoption of new technologies are essential, and public extension is a main channel for spreading them \citep{takahashi_technology_2020}.
		\vspace{0.35cm}
		\item Yet public extension often delivers poorly: agents are absent and visit farmers rarely \citep{anderson_chapter_2007, alam_incentives_2024}.
		\vspace{0.35cm}
		\item \textbf{How can extension agents be motivated to deliver better services?}
	\end{itemize}
\end{frame}

\begin{frame}{Motivation: Incentives for Public Service Providers}
	\begin{itemize}\small
		\item Providers' effort is hard to observe, so it is only loosely linked to rewards \citep{chaudhury_missing_2006, das_money_2007, finan_chapter_2017}.
		\vspace{0.2cm}
		\item Pay, monitoring and recognition can improve performance \citep{muralidharan_teacher_2011, duflo_incentives_2012, ashraf_awards_2014, mansuri_money_2023}.
		\vspace{0.2cm}
		\item Emerging literature focusing on \textbf{belief that one's work matters}:
		\begin{itemize}\small
			\item \citet{Kaur2023}: teachers in India --- a psycho-social intervention on teachers' sense of control raised classroom effort (+0.13 SD) and student learning (+0.09 SD).
			\item \citet{khan_mission_2025}: community health workers in Pakistan --- training that strengthened mission motivation increased household visits and other tasks, and improved child health.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Motivation: Incentives for Extension Agents}
	\begin{itemize}\small
		\item Few experiments target \textbf{public extension agents}:
		\begin{itemize}\small
			\item \citet{alam_incentives_2024}: Bangladesh, 807 officers in 40 sub-district offices --- a one-time cash reward and recognition by the district director both raised (self-reported) performance by about 1 SD.
			\item \citet{Genicot2022}: Punjab, Pakistan --- pay-for-performance tied to a digital monitoring system raised outreach; only the scheme adding top-down monitoring improved quality as reported by farmers.
		\end{itemize}
		\vspace{0.2cm}
		\item Both rely on money, supervisors or monitoring. 
		\vspace{0.2cm}
        \item Yet agents themselves say that farmers' gratitude
		motivates them \citep{Ragasa2013}. 
	\end{itemize}
\end{frame}

\begin{frame}{Research Question and What We Do}
	\begin{center}
	\textbf{RQ: Can non-monetary interventions that enhance a sense that their work matters change agents' behaviour? Do they improve the farm-level outcome?}
	\end{center}
	\vspace{0.2cm}
	\begin{itemize}\small
		\item \textbf{Cluster RCT} with Ghana's public extension service: 42 Agricultural Extension Agents (AEA) in 8 rainfed rice districts, 378 sampled farmers; randomised at the \textbf{district} level.
		\begin{itemize}
			\item \textbf{T1 --- Feedback}: bi-weekly reports of farmers' assessments, sent to the AEA and the supervisor.
			\item \textbf{T2 --- Feedback + Training}: plus a sensitization training on the social value of extension.
			\item \textbf{Cross-cutting --- GPS stamp}: half of the AEAs required to submit their visiting records with a GPS form.
		\end{itemize}
	\vspace{0.2cm}
		\item Baseline, Endline, and \textbf{Bi-weekly farmer phone survey} (11 rounds, Apr--Sep 2025): effort measured by farmers AEA serves.
	\end{itemize}
\end{frame}

\begin{frame}{Preview of Findings}
	\begin{enumerate}\small
		\item \textbf{Service delivery}: in the planting season, farmers in T1 and T2 received 0.22 and 0.27 more visits per two weeks (control 0.77); over the whole season only T2. Both arms raised satisfaction and the share of helpful advice.
		\vspace{0.12cm}
		\item \textbf{AEA}: no change in intrinsic, extrinsic or prosocial motivation, but a stronger locus of control (T1: general; T1 and T2: agriculture). Training raised knowledge and job satisfaction.
		\vspace{0.12cm}
		\item \textbf{Farms}: no effect on yield or income; less seed and herbicide and a shift to hired labour; profit rises when family labour is valued at the hired wage. The GPS stamp had no effect.
		\vspace{0.12cm}
		\item \textbf{Measurement}: agents' self-reported visits are uncorrelated with farmers' reports.
	\end{enumerate}
	\vspace{0.1cm}
	{\footnotesize With 8 district clusters, several estimates are significant under randomization inference but not under the wild cluster bootstrap.}
\end{frame}

\begin{frame}{Contributions}
	\begin{itemize}\small
		\item \textbf{Agricultural extension}: most experiments study training and targeting of farmers \citep{benyishay_social_2019, Beaman2021} or incentives for lead farmers \citep{shikuku_incentives_2019, balew_incentivizing_2023, okello_social_2023}.
        The few on public extension agents use money or monitoring \citep{alam_incentives_2024, Genicot2022}. \\
        $\to$ Our feedback comes from the farmers served and involves no reward.
		\vspace{0.2cm}
		\item \textbf{Measuring agent effort}: studies rely on self-reports or administrative records
		\citep[e.g.,][]{alam_incentives_2024}. \\
        $\to$ Bi-weekly farmer surveys give an independent measure of visits, and they are uncorrelated with agents' self-reports.
		\vspace{0.2cm}
		\item \textbf{Motivation of public service providers}: incentives and beliefs shape teachers' and health workers' effort \citep{duflo_incentives_2012, ashraf_awards_2014, banuri_pro-social_2016, Kaur2023, khan_mission_2025}, but few studies measure motivation and beliefs alongside behaviour and beneficiary outcomes. \\
        $\to$ We measure changes in belief and motivation as well as behaviours and beneficiary outcomes.
	\end{itemize}
\end{frame}

\begin{frame}{Agricultural extension in Ghana}
	\begin{itemize}
		\item Agriculture accounts for 21.3\% of GDP and 34.6\% of employment
		\citep{worldbank_wdi_2026}; domestic rice production meets only 55.1\% of demand
		\citep{ndpc_progress_2025}.
		\vspace{0.3cm}
		\item Extension is delivered by MoFA district offices, with more than 1{,}500 farmers
		per agent against a benchmark of 500 \citep{MoFA2025Extension}; agents lack funds for
		motorcycle fuel, so distant communities are rarely visited.
		\vspace{0.3cm}
		\item JICA and MoFA's GRIP (2022--2027) trains agents in improved rice practices
		\citep{jica_grip_2022}; we ask whether non-monetary incentives add to this.
	\end{itemize}
\end{frame}

\begin{frame}{Interventions}
	\begin{itemize}\footnotesize
		\item \textbf{Feedback (T1 and T2)}: a bi-weekly phone survey of all sample farmers in all arms (visits,
		contacts, satisfaction 1--5 and its reasons). Every two weeks, AEAs in T1 and T2 and their supervisors
		received a one-page report: own visits per farmer against the previous round, the district average and
		the best in the district; satisfaction ratings; farmers' reasons verbatim, without names.
		\vspace{0.15cm}
		\item \textbf{Training (T2)}: overnight sensitization training, 29 April 2025, Koforidua; delivered by two
		assistant directors of MoFA's DAES. No technical content --- the social value of extension and how agents'
		own efforts improve farmers' livelihoods. All 15 invited AEAs attended.
		\vspace{0.15cm}
		\item \textbf{GPS stamp (cross-cutting)}: 20 of 42 AEAs asked to record each farm visit with a KoboToolbox
		form that captures GPS coordinates. Not shared with supervisors; no reward or penalty. 19 of 20 AEAs
		submitted 357 records.
		\vspace{0.15cm}
		\item \textbf{Fuel support (all arms)}: 150 GHS per month by mobile money, Apr--Aug 2025, replacing a
		phased-out project allowance; common to all arms.
	\end{itemize}
\end{frame}

\begin{frame}{Empirical Strategy: ANCOVA}
	\begin{itemize}\small
		\item \textbf{AEA level} ($N=41$):
		\[ A_{jd1} = \beta_0 + \beta_1 T1_d + \beta_2 T2_d + \beta_3 GPS_j + \zeta A_{jd0} + \mu_r + \varepsilon_{jd} \]
		\item \textbf{Farm level} ($N=373$):
		\[ F_{ijd1} = \gamma_0 + \gamma_1 T1_d + \gamma_2 T2_d + \gamma_3 GPS_j + \eta F_{ijd0} + \mu_r + \varepsilon_{ijd} \]
		\item ANCOVA rather than DID: more precise when outcomes are weakly autocorrelated, as farm income and profit are \citep{mckenzie_beyond_2012}.
		\item SE clustered by district (8 clusters); \textbf{wild cluster bootstrap} (Webb, 9{,}999) and \textbf{randomization inference} (1{,}000) $p$-values.
	\end{itemize}
\end{frame}

\renewcommand{\footsec}{5.~Conclusion}
\begin{frame}[plain]
	\vfill
	\centering
	{\Huge\bfseries\textcolor{myGreen}{5.~Conclusion}}
	\vfill
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
	\item \textbf{What we find}: feedback, alone or with training, raised visits in the planting
	season, farmers' satisfaction and the share of helpful advice, and strengthened agents' locus
	of control but not their motivation. Farms: no change in yield or income, but less seed and
	herbicide and a shift to hired labour. The GPS stamp had no effect.
	\vspace{0.3cm}
	\item \textbf{Policy implications}: agents' self-reported visits are uncorrelated with
	farmers' reports, so effort needs independent verification --- farmer phone surveys are one
	low-cost option; non-monetary measures can improve service delivery.
	\vspace{0.3cm}
	\item \textbf{Caveats}: 8 district clusters (2 control), so several results hold under
	randomization inference but not the wild bootstrap; one season only; the mechanism is suggestive.
\end{itemize}
\end{frame}

\begin{frame}{Motivation: Public Service and Incentives}
	\begin{itemize}
		\item Despite its importance for development, public workers in low-income
		countries perform poorly \citep{duflo_incentives_2012, finan_chapter_2017}.
		\begin{itemize}
			\item \textbf{Quantity}: across six countries, 19\% of teachers and 35\% of health
			workers were absent from their posts \citep{chaudhury_missing_2006}.
			\item \textbf{Quality}: doctors in India ``do less than what they know they should''
			\citep{das_money_2007}.
		\end{itemize}
		\item Incentives can improve performance.
		\begin{itemize}
			\item \textbf{Monetary incentives} raised teachers' effort and students' learning
			\citep{leaver_recruitment_2021, muralidharan_teacher_2011}.
			\item \textbf{Monitoring} reduced absence among teachers \citep{duflo_incentives_2012}
			and nurses \citep{banerjee_putting_2008}.
			\item \textbf{Social recognition} improved the performance of health workers
			\citep{ashraf_awards_2014} and village leaders \citep{mansuri_money_2023}.
			\item \textbf{Mission-motivation training} improved community health workers'
			performance \citep{khan_mission_2025}; strengthening teachers' belief that their
			effort matters raised their effort and student learning \citep{Kaur2023}.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Motivation: Agricultural Extension}
	\begin{itemize}
		%\item Raising agricultural productivity requires farmers to adopt new technologies and improved management practices \citep{takahashi_technology_2020}.
		\item Extension agents are the main channel through which new technologies and
		improved practices reach farmers, \textbf{but} agent absenteeism and weak service
		delivery are widely reported \citep{anderson_chapter_2007, alam_incentives_2024}.
		\item What incentive evidence exists in agriculture is mostly about \textbf{lead farmers
		and volunteer trainers}, not salaried public agents \citep{balew_incentivizing_2023,
		benyishay_social_2019, shikuku_incentives_2019, okello_social_2023}.
		\item Only two studies have tested incentives for extension agents themselves:
		\begin{itemize}
			\item \citet{alam_incentives_2024}: Bangladesh, 40 subdistrict offices --- a
			\textbf{cash reward} versus \textbf{recognition by the district director}.
			\item \citet{Genicot2022}: Punjab, Pakistan --- three
			\textbf{pay-for-performance} schemes tied to a digital monitoring system.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{What We Still Do Not Know}
	\begin{itemize}\small
		\item Existing evidence on extension workers speaks to \textbf{extrinsic} incentives: money, monitoring, and recognition granted by supervisors.

		\vspace{0.18cm}
		\item \textbf{Can recognition come from below?} Farmers observe an agent's effort
		directly, yet their voice is rarely fed back to the agent, even though agents
		themselves rank the absence of appreciation
		from farmers among the main reasons for their low motivation \citep{Ragasa2013}. Such a scheme also costs less than monitoring or pay.

		\vspace{0.18cm}
		\item \textbf{Can the belief that one's effort matters be moved?} If agents doubt that
		their work changes farmers' lives, a weak \emph{locus of control}, then
		pay-for-performance is addressing the wrong constraint.

		\vspace{0.18cm}
		\item \textbf{Does better service reach the farm?} Prior studies stop at the agent's
		outreach, not at adoption, yields and income.

		\vspace{0.18cm}
		\item \textbf{Effort is measured by those who exert it}: administrative records or the
		agent's own report.
	\end{itemize}
\end{frame}
