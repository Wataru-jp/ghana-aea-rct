\begin{frame}{Motivation: Public Service and Incentives}
	\begin{itemize}
		\item Despite its importance for development, public service delivery in low-income
		countries performs poorly \citep{finan_chapter_2017}.
		\begin{itemize}
			\item \textbf{Quantity}: across six countries, 19\% of teachers and 35\% of health
			workers were absent from their posts \citep{chaudhury_missing_2006}.
			\item \textbf{Quality}: doctors in India ``do less than what they know they should''
			\citep{das_money_2007}.
		\end{itemize}
		\item Incentives can improve performance.
		\begin{itemize}
			\item \textbf{Monetary incentives} raised teachers' effort and students' learning
			\citep{leaver_recruitment_2021, muralidharan_teacher_2011}.
			\item \textbf{Monitoring} reduced absence among teachers \citep{duflo_incentives_2012}
			and nurses \citep{banerjee_putting_2008}.
			\item \textbf{Social recognition} improved the performance of health workers
			\citep{ashraf_awards_2014} and village leaders \citep{mansuri_money_2023}.
			\item Strengthening \textbf{locus of control} increased community health workers'
			delivery of both core and additional tasks \citep{khan_mission_2025}.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Motivation: Agricultural Extension}
	\begin{itemize}
		\item Raising agricultural productivity requires farmers to adopt new technologies and
		improved management practices \citep{takahashi_technology_2020}.
		\item Agricultural extension agents (AEAs) are the main channel through which that
		knowledge reaches farmers \citep{anderson_chapter_2007}.
		\item \textbf{But} AEA absenteeism and weak service delivery are widely reported
		\citep{anderson_chapter_2007, alam_incentives_2024}.
		\item Two studies have tested incentives for extension agents:
		\begin{itemize}
			\item \citet{alam_incentives_2024}: Bangladesh, 807 officers --- a \textbf{cash
			reward} versus \textbf{recognition by the district director}.
			\item \citet{genicot_carrots_nodate}: Punjab, Pakistan --- three
			\textbf{pay-for-performance} schemes tied to a digital monitoring system.
		\end{itemize}
		\item Both reward the agent \textbf{from above}, through pay or through supervisors.
	\end{itemize}
\end{frame}

\begin{frame}{What We Still Do Not Know}
	\begin{itemize}\small
		\item Existing evidence speaks to \textbf{extrinsic} incentives: money, and recognition
		granted by supervisors.

		\vspace{0.18cm}
		\item \textbf{Can recognition come from below?} Farmers observe an agent's effort
		directly, yet their voice is rarely fed back to the agent --- and such a scheme costs
		almost nothing, which matters where civil-service pay is fixed by statute.

		\vspace{0.18cm}
		\item \textbf{Can the belief that one's effort matters be moved?} If agents doubt that
		their work changes farmers' lives --- a weak \emph{locus of control} --- then
		pay-for-performance is addressing the wrong constraint. Whether training can shift that
		belief, and with it behaviour, is untested in extension.

		\vspace{0.18cm}
		\item \textbf{Does better service reach the farm?} Prior studies stop at the agent's
		outreach, not at adoption, yields and income.

		\vspace{0.18cm}
		\item \textbf{Measurement is itself part of the problem}: effort is normally taken from
		administrative records or the agent's own report --- both of which the agent controls.
	\end{itemize}
\end{frame}

\begin{frame}{Research Questions}
	\begin{enumerate}
		\item Does \textbf{feedback from farmers} --- telling an agent how satisfied the farmers
		he or she serves are --- raise extension effort and the quality of the advice given?

		\vspace{0.35cm}
		\item Does \textbf{training on the social significance} of extension work, aimed at the
		agent's locus of control, add anything on top of that feedback?

		\vspace{0.35cm}
		\item Do changes at the agent level reach the \textbf{farm level} --- management
		practices, input use, yields and income?
	\end{enumerate}
\end{frame}

\begin{frame}{What We Do}
	\begin{itemize}\small
		\item A \textbf{cluster-randomised controlled trial} with Ghana's public extension
		service: 44 AEAs across 10 districts, serving about 390 rice farmers. Randomisation is
		at the \textbf{district} level.

		\vspace{0.18cm}
		\item \textbf{T1 --- Feedback}: farmers' satisfaction with their AEA is collected and
		reported back to the agent. Recognition from below, at essentially no cost.
		\item \textbf{T2 --- Feedback + Training}: the same feedback, plus training that
		emphasises how extension work matters for farmers' livelihoods.
		\item \textbf{Cross-cutting --- GPS stamp}: agents record their location when they visit
		fields; a light form of monitoring.

		\vspace{0.18cm}
		\item \textbf{How we measure effort}: besides baseline and endline surveys, a
		\textbf{bi-weekly farmer survey} over 11 rounds (Apr--Sep 2025) --- effort is reported by
		those who receive the service, not by the agent.
		\item We keep \textbf{intrinsic motivation}, \textbf{prosocialness} and \textbf{locus of
		control} as separate constructs rather than one ``motivation'' index: they are distinct,
		and the data show they move differently.
	\end{itemize}
\end{frame}

\begin{frame}{Empirical Strategy}
\begin{itemize}
	\item \textbf{ANCOVA}: regress endline outcome on treatment status, controlling for
	      the baseline value of the outcome
	\item Arms: T1 (Feedback), T2 (Feedback + Training); plus the cross-cutting Stamp (GPS)
	\item \textbf{Sample}: \emph{rainfed sites only} --- the two irrigation schemes (Kpong and
	      Weta) are dropped from every estimation (12 farmers, 2 AEAs, all in the control arm;
	      yields, seed rates and labour there are on a different scale). Farmers $N=382$, AEAs $N=41$
	\item Farmer models add region (province) FE; SEs clustered at district
	      (\textbf{8 clusters}: 2 control, 3 T1, 3 T2)
	\item Few clusters $\Rightarrow$ \textbf{wild cluster bootstrap} (Webb, 9{,}999) and
	      \textbf{randomization-inference} $p$-values alongside clustered SEs. With 2 control
	      districts these bind --- read the clustered SEs as the optimistic bound
	\item Crop failure = realized \textbf{zero} output (kept in sample); logs use $\ln(1+x)$;
	      outcomes are not winsorized
\end{itemize}
\end{frame}
